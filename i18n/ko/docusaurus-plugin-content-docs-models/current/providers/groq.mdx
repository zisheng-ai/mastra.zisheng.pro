}

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "groq/whisper-large-v3-turbo"
      : "groq/allam-2-7b";
  }
});
