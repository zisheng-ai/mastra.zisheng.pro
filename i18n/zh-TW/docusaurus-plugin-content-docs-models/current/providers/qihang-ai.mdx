const agent = new Agent({
  id: 'dynamic-agent',
  name: 'Dynamic Agent',
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === 'complex'
    return useAdvanced ? 'qihang-ai/gpt-5.2-codex' : 'qihang-ai/claude-haiku-4-5-20251001'
  },
})
