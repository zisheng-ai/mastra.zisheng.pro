  id: 'dynamic-agent',
  name: 'Dynamic Agent',
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === 'complex'
    return useAdvanced ? 'perplexity/sonar-reasoning-pro' : 'perplexity/sonar'
  },
})
