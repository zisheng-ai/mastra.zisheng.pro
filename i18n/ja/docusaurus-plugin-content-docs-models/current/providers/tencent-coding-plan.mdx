const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "tencent-coding-plan/tc-code-latest"
      : "tencent-coding-plan/glm-5";
  }
});
