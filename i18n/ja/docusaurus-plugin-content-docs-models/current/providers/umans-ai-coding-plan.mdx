
```typescript title="src/mastra/agents/my-agent.ts" {7}
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "my-agent",
  name: "My Agent",
  instructions: "You are a helpful assistant",
  model: "umans-ai-coding-plan/umans-coder"
});

// Generate a response
const response = await agent.generate("Hello!");

// Stream a response
const stream = await agent.stream("Tell me a story");
for await (const chunk of stream) {
  console.log(chunk);
}
```

## 高度な設定

### カスタムヘッダー

```typescript title="src/mastra/agents/my-agent.ts"
const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://api.code.umans.ai/v1",
    id: "umans-ai-coding-plan/umans-coder",
    apiKey: process.env.UMANS_AI_CODING_PLAN_API_KEY,
    headers: {
      "X-Custom-Header": "value"
    }
  }
});
```

### 動的なモデル選択

```typescript title="src/mastra/agents/my-agent.ts"
const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "umans-ai-coding-plan/umans-qwen3.6-35b-a3b"
      : "umans-ai-coding-plan/umans-coder";
  }
});
```
