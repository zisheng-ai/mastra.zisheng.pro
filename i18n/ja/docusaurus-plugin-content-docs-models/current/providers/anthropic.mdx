import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "my-agent",
  name: "My Agent",
  instructions: "You are a helpful assistant",
  model: "anthropic/claude-fable-5"
});

// Generate a response
const response = await agent.generate("Hello!");

// Stream a response
const stream = await agent.stream("Tell me a story");
for await (const chunk of stream) {
  console.log(chunk);
}

## 高度な設定

### カスタムヘッダー

```typescript title="src/mastra/agents/my-agent.ts"
const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    id: "anthropic/claude-fable-5",
    apiKey: process.env.ANTHROPIC_API_KEY,
    headers: {
      "X-Custom-Header": "value"
    }
  }
});
```

### 動的なモデル選択

```typescript title="src/mastra/agents/my-agent.ts"
const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "anthropic/claude-sonnet-5"
      : "anthropic/claude-fable-5";
  }
});
```

## Provider を直接インストールする

この Provider はスタンドアロン package として直接インストールし、Mastra のモデルルーター文字列の代わりに使用することもできます。詳しくは、[package のドキュメント](https://www.npmjs.com/package/@ai-sdk/anthropic)を参照してください。

```bash npm2yarn
npm install @ai-sdk/anthropic
```

Provider 固有の詳細については、[AI SDK Anthropic Provider のドキュメント](https://ai-sdk.dev/providers/ai-sdk-providers/anthropic)を参照してください。
