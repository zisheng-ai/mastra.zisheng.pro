  id: 'dynamic-agent',
  name: 'Dynamic Agent',
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === 'complex'
    return useAdvanced ? 'anthropic/claude-sonnet-5' : 'anthropic/claude-fable-5'
  },
})
