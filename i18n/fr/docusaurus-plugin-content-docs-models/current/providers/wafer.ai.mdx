    id: "wafer.ai/GLM-5.1",

### Sélection dynamique du modèle

```typescript title="src/mastra/agents/my-agent.ts"
const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "wafer.ai/glm5.2-fast"
      : "wafer.ai/GLM-5.1";
  }
});
```
