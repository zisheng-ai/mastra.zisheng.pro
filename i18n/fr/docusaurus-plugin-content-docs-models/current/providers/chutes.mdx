const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "chutes/zai-org/GLM-5.2-TEE"
      : "chutes/Nemotron-3-Nano-Omni-30B-TEE";
  }
});
